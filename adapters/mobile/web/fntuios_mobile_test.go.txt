package main

import (
	"testing"

	"github.com/Gaurav-Gosain/sip"
	"github.com/Gaurav-Gosain/tuios/internal/config"
)

func TestFnMobileControlsFollowConfiguredSidebarKey(t *testing.T) {
	user := config.DefaultConfig()
	user.Keybindings.PrefixMode["prefix_explore"] = []string{"ctrl+e"}
	registry := config.NewKeybindRegistry(user)
	prefix, rows := mobileBar(registry, "ctrl+a")
	cfg := sip.Config{MobilePrefix: prefix, MobileRows: rows}
	installFnMobileControls(&cfg, registry, "ctrl+a")
	key, ok := findKey(cfg.MobileRows[0].Keys, "侧栏")
	if !ok || key.Key != "e" || !key.Ctrl || !key.Prefixed {
		t.Fatalf("sidebar button ignored configured binding: %+v", key)
	}
	if cfg.MobilePrefix.Key != "a" || !cfg.MobilePrefix.Ctrl {
		t.Fatal("mobile commands changed the configured leader")
	}
	if _, ok := findKey(cfg.MobileRows[0].Keys, "会话"); !ok {
		t.Fatal("session picker is not reachable from the touch row")
	}
}

func TestFnMobileControlsDoNotInventUnboundActions(t *testing.T) {
	user := config.DefaultConfig()
	user.Keybindings.PrefixMode["prefix_explore"] = []string{"f4"}
	registry := config.NewKeybindRegistry(user)
	prefix, rows := mobileBar(registry, "ctrl+b")
	cfg := sip.Config{MobilePrefix: prefix, MobileRows: rows}
	installFnMobileControls(&cfg, registry, "ctrl+b")
	if _, ok := findKey(cfg.MobileRows[0].Keys, "侧栏"); ok {
		t.Fatal("sidebar button retained an unencodable binding")
	}
	prefix, rows = mobileBar(registry, "f1")
	cfg = sip.Config{MobilePrefix: prefix, MobileRows: rows}
	installFnMobileControls(&cfg, registry, "f1")
	if len(cfg.MobileRows) != 1 || !cfg.MobilePrefix.IsZero() {
		t.Fatal("an unavailable leader produced unsafe command buttons")
	}
}
